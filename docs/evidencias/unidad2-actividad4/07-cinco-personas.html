SELECT u.name,cu.id AS membresia,cu.company_id,cu.is_admin,r.code AS rol FROM company_users cu JOIN users u ON u.id=cu.user_id JOIN company_user_roles cr ON cr.company_user_id=cu.id AND cr.is_active=1 JOIN roles r ON r.id=cr.role_id WHERE cu.company_id=1 ORDER BY cu.id
